# Item Gate Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add the Item Gate: a barrel-style, 27-slot block that drops one stack every 8 ticks out of its output face, unloading shulker boxes stack by stack and then dropping the empty box.

**Architecture:** Own `ItemGateBlock` (6-way `FACING`, `OPEN`, `ENABLED`) plus `ItemGateBlockEntity` extending vanilla `RandomizableContainerBlockEntity` (barrel pattern: opener counter drives `OPEN`, menu is vanilla `ChestMenu.threeRows`, so no screen code). Shulker handling is a pure helper `ShulkerUnloader` over the `CONTAINER` data component. Drop cycle ticks on the server with an 8-tick cooldown. Verified with Fabric GameTests, including an end-to-end test that feeds a Sorter Hopper.

**Tech Stack:** Java 25, Minecraft 26.2 (Mojang names), Fabric Loader 0.19.5, Fabric API 0.161.0+26.2, Fabric GameTest API.

**Spec:** `docs/superpowers/specs/2026-09-23-item-gate-design.md`

## Global Constraints

- Minecraft `~26.2`, Java `>=25`. Every Gradle command runs with `export JAVA_HOME=/opt/homebrew/opt/openjdk@25` (system `java` is 8).
- Mod id `easystorage`; package `tr.gokhan.easystorage`; ids via `EasyStorage.id(path)`.
- No mixins, no access wideners.
- Drop cadence: one stack per `HopperBlockEntity.MOVE_ITEM_SPEED` (8) ticks; drops are exactly 8 ticks apart.
- 27 slots; screen is vanilla `ChestMenu.threeRows` (no custom menu/screen).
- Only items in `ItemTags.SHULKER_BOXES` are unloaded; everything else drops as-is.
- Drop spawns 0.3 blocks outside the output face centre, zero velocity, no sound/particles.
- Recipe: shapeless barrel + dropper + hopper + repeater → 1 `easystorage:item_gate`; unlocked by obtaining a dropper.
- Do not copy Mojang textures; placeholders are generated by `tools/placeholder_textures.py`.
- Existing Sorter Hopper behaviour and its 10 GameTests must keep passing.

## Review Focus

Failure modes the spec implies that no automated test pins; the reviewer and the manual check should look at these:

1. **Output face pointing up:** dropped stacks spawn above the gate and fall back onto its top; expected: they rest there (or roll off) without errors, and nothing duplicates. Covered by manual check step 5.
2. **World reload mid-unload:** a half-unloaded shulker box and the drop cooldown persist; expected: unloading resumes after reload with no loss. Covered by manual check step 6.
3. **Player removes a shulker box mid-unload:** the player gets the box with its remaining contents; expected by design (the box never leaves its slot until empty). Covered by manual check step 4.
4. **Hopper on the output side:** it pulls items out of the gate (standard container behaviour, deliberately not blocked by the spec). Reviewer: confirm nothing in the code tries to block it.
5. **Dyed shulker boxes:** unloaded exactly like plain ones (all 17 are in `ItemTags.SHULKER_BOXES`). Pinned by the red-shulker assertion in Task 1's test.

```java
package tr.gokhan.easystorage.block.entity;

import java.util.ArrayList;
import java.util.List;
import net.minecraft.core.component.DataComponents;
import net.minecraft.tags.ItemTags;
import net.minecraft.world.item.ItemStack;
import net.minecraft.world.item.component.ItemContainerContents;

/**
 * Takes stored stacks out of a shulker box item one at a time. Pure item logic,
 * no world access.
 */
public final class ShulkerUnloader {
	private ShulkerUnloader() {
	}

	/** True when {@code stack} is a shulker box that still stores at least one item. */
	public static boolean hasContents(ItemStack stack) {
		return stack.is(ItemTags.SHULKER_BOXES)
			&& stack.getOrDefault(DataComponents.CONTAINER, ItemContainerContents.EMPTY).nonEmptyItems().iterator().hasNext();
	}

	/**
	 * Removes the first non-empty stored stack from {@code shulker} and returns it.
	 * Returns {@link ItemStack#EMPTY} when nothing is stored.
	 */
	public static ItemStack takeNext(ItemStack shulker) {
		ItemContainerContents contents = shulker.getOrDefault(DataComponents.CONTAINER, ItemContainerContents.EMPTY);
		List<ItemStack> items = new ArrayList<>(contents.allItemsCopyStream().toList());
		for (int i = 0; i < items.size(); i++) {
			if (!items.get(i).isEmpty()) {
				ItemStack taken = items.set(i, ItemStack.EMPTY);
				shulker.set(DataComponents.CONTAINER, ItemContainerContents.fromItems(items));
				return taken;
			}
		}
		return ItemStack.EMPTY;
	}
}
```

```java
		}
```

```java
		this.items = NonNullList.withSize(this.getContainerSize(), ItemStack.EMPTY);
		if (!this.tryLoadLootTable(input)) {
			ContainerHelper.loadAllItems(input, this.items);
		}
		this.cooldownTime = input.getIntOr("DropCooldown", 0);
	}

	@Override
	protected void saveAdditional(ValueOutput output) {
		super.saveAdditional(output);
		if (!this.trySaveLootTable(output)) {
			ContainerHelper.saveAllItems(output, this.items);
		}
		output.putInt("DropCooldown", this.cooldownTime);
	}

	@Override
	public int getContainerSize() {
		return CONTAINER_SIZE;
	}

	@Override
	protected NonNullList<ItemStack> getItems() {
		return this.items;
	}

	@Override
	protected void setItems(NonNullList<ItemStack> items) {
		this.items = items;
	}

	@Override
	protected Component getDefaultName() {
		return DEFAULT_NAME;
	}

	@Override
	protected AbstractContainerMenu createMenu(int containerId, Inventory inventory) {
		return ChestMenu.threeRows(containerId, inventory, this);
	}

	// --- Opening (drives the OPEN block state, like the vanilla barrel) ---

	@Override
	public void startOpen(ContainerUser user) {
		if (!this.remove && !user.getLivingEntity().isSpectator()) {
			this.openersCounter.incrementOpeners(user.getLivingEntity(), this.getLevel(), this.getBlockPos(), this.getBlockState(), user.getContainerInteractionRange());
		}
	}

	@Override
	public void stopOpen(ContainerUser user) {
		if (!this.remove && !user.getLivingEntity().isSpectator()) {
			this.openersCounter.decrementOpeners(user.getLivingEntity(), this.getLevel(), this.getBlockPos(), this.getBlockState());
		}
	}

	@Override
	public List<ContainerUser> getEntitiesWithContainerOpen() {
		return this.openersCounter.getEntitiesWithContainerOpen(this.getLevel(), this.getBlockPos());
	}

	public void recheckOpen() {
		if (!this.remove) {
			this.openersCounter.recheckOpeners(this.getLevel(), this.getBlockPos(), this.getBlockState());
		}
	}

	private void setOpen(BlockState state, boolean open) {
		this.level.setBlock(this.getBlockPos(), state.setValue(ItemGateBlock.OPEN, open), Block.UPDATE_ALL);
	}

	// --- Dropping ---

	public static void dropTick(Level level, BlockPos pos, BlockState state, ItemGateBlockEntity gate) {
		if (gate.cooldownTime > 0) {
			gate.cooldownTime--;
			return;
		}
		if (!state.getValue(ItemGateBlock.ENABLED) || gate.isEmpty()) {
			return;
		}
		if (gate.dropNext(level, pos, state.getValue(ItemGateBlock.FACING))) {
			gate.cooldownTime = HopperBlockEntity.MOVE_ITEM_SPEED - 1;
			gate.setChanged();
		}
	}

	/** Drops one stack out of the {@code facing} face. Returns false when nothing was dropped. */
	private boolean dropNext(Level level, BlockPos pos, Direction facing) {
		BlockPos target = pos.relative(facing);
		if (level.getBlockState(target).isCollisionShapeFullBlock(level, target)) {
			return false;
		}

		ItemStack toDrop = this.takeNextStack();
		if (toDrop.isEmpty()) {
			return false;
		}

		double x = pos.getX() + 0.5 + facing.getStepX() * DROP_OFFSET;
		double y = pos.getY() + 0.5 + facing.getStepY() * DROP_OFFSET;
		double z = pos.getZ() + 0.5 + facing.getStepZ() * DROP_OFFSET;
		// Item entities are positioned by their feet; lower by half their height to centre them.
		ItemEntity entity = new ItemEntity(level, x, y - 0.125, z, toDrop, 0.0, 0.0, 0.0);
		level.addFreshEntity(entity);
		return true;
	}

	/** Removes the next stack to drop: a shulker box's next stored stack, or the whole slot. */
	private ItemStack takeNextStack() {
		for (int slot = 0; slot < this.getContainerSize(); slot++) {
			ItemStack stack = this.getItem(slot);
			if (stack.isEmpty()) {
				continue;
			}
			if (ShulkerUnloader.hasContents(stack)) {
				return ShulkerUnloader.takeNext(stack);
			}
			return this.removeItemNoUpdate(slot);
		}
		return ItemStack.EMPTY;
	}
}
```

```python
#!/usr/bin/env python3
"""Generate 16x16 placeholder block textures for Easy Storage.

Pure standard library (no Pillow). Output is deterministic, so re-running it
produces identical files. Final art will replace these.

Usage: python3 tools/placeholder_textures.py
"""
import struct
import zlib
from pathlib import Path

OUT = Path(__file__).resolve().parent.parent / "src/main/resources/assets/easystorage/textures/block"

# Blue-tinted iron, matching the filter-slot tint used in the screen (0x3A6EA5).
BASE = (0x3A, 0x6E, 0xA5)
LIGHT = (0x6F, 0x9C, 0xCC)
DARK = (0x22, 0x3F, 0x60)

# Item Gate: dark teal barrel wood; orange marks the output face (mod-wide convention).
TEAL = (0x2B, 0x6B, 0x66)
TEAL_LIGHT = (0x4F, 0x94, 0x8C)
TEAL_DARK = (0x17, 0x40, 0x3D)
ORANGE = (0xE0, 0x7A, 0x1F)
HOLE = (0x10, 0x10, 0x10)
HOLE_OPEN = (0x5A, 0x3A, 0x1E)


def png(path, pixels):
    raw = b"".join(b"\x00" + bytes(c for px in row for c in px) for row in pixels)

    def chunk(tag, data):
        body = tag + data
        return struct.pack(">I", len(data)) + body + struct.pack(">I", zlib.crc32(body) & 0xFFFFFFFF)

    data = (
        b"\x89PNG\r\n\x1a\n"
        + chunk(b"IHDR", struct.pack(">IIBBBBB", 16, 16, 8, 6, 0, 0, 0))
        + chunk(b"IDAT", zlib.compress(raw, 9))
        + chunk(b"IEND", b"")
    )
    path.write_bytes(data)


def shade(rgb, amount):
    return tuple(max(0, min(255, c + amount)) for c in rgb)


def noise(x, y, seed):
    n = (x * 374761393 + y * 668265263 + seed * 1442695041) & 0xFFFFFFFF
    n = (n ^ (n >> 13)) * 1274126177 & 0xFFFFFFFF
    return (n & 0xF) - 8  # -8..7


def texture(seed, fill, edge_top, edge_bottom):
    rows = []
    for y in range(16):
        row = []
        for x in range(16):
            if y == 0 or x == 0:
                rgb = edge_top
            elif y == 15 or x == 15:
                rgb = edge_bottom
            else:
                rgb = shade(fill, noise(x, y, seed))
            row.append(rgb + (255,))
        rows.append(row)
    return rows


def staves(seed):
    """Barrel-style side: vertical planks with a dark band near each end."""
    rows = []
    for y in range(16):
        row = []
        for x in range(16):
            if y in (2, 13):
                rgb = TEAL_DARK
            elif x % 4 == 0:
                rgb = shade(TEAL, -25)
            else:
                rgb = shade(TEAL, noise(x, y, seed))
            row.append(rgb + (255,))
        rows.append(row)
    return rows


def face(seed, center):
    """Barrel end: teal rim; centre is an orange-ringed hole, or plain lid when center is None."""
    rows = []
    for y in range(16):
        row = []
        for x in range(16):
            edge = x in (0, 15) or y in (0, 15)
            if edge:
                rgb = TEAL_DARK
            elif center is not None and 5 <= x <= 10 and 5 <= y <= 10:
                ring = x in (5, 10) or y in (5, 10)
                rgb = ORANGE if ring else center
            else:
                rgb = shade(TEAL_LIGHT, noise(x, y, seed))
            row.append(rgb + (255,))
        rows.append(row)
    return rows
```
